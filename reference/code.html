import { FacturaFlowClient } from '@facturaflow/sdk';
import fs from 'fs';

const client = new FacturaFlowClient({
  apiKey: process.env.FACTURAFLOW_API_KEY!,
});

async function processInvoice() {
  const fileStream = fs.createReadStream('./fatura-telecom.pdf');

  // Synchronous inference under ~400ms
  const result = await client.invoices.extract({
    file: fileStream,
    autoValidateTaxes: true,
    outputFormat: 'json_normalized',
  });

  console.log(`Fatura Processada: ${result.invoiceNumber}`);
  console.log(`Total Líquido: R$ ${result.financials.netPayable}`);
  console.log(`Score de Auditoria: ${result.auditConfidence * 100}%`);
}

Perguntas Frequentes

Todas as faturas trafegam sob criptografia TLS 1.3 de ponta a ponta e são armazenadas em repouso com padrão militar AES-256 em data centers soberanos no Brasil (AWS São Paulo). Não utilizamos dados de faturas dos clientes para re-treinar modelos de IA públicos. Cumprimos integralmente a LGPD e os protocolos SOC 2 Type II.

O FacturaFlow aceita arquivos digitais nativos e escaneados nos formatos: PDF (pesquisável ou rasterizado), TIFF multipágina, PNG, JPEG e XMLs de NFe/CTe. Nossos algoritmos aplicam desinclinação e pré-processamento de ruído automaticamente antes da extração.

Para arquivos de até 3 páginas, a extração síncrona tem latência média de 420ms (com p99 abaixo de 1.1s). Para lotes contendo centenas de arquivos enviados simultaneamente, os lotes são despachados para filas paralelas e os resultados são entregues de modo assíncrono via Webhooks.
